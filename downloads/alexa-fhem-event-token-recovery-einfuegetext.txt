Server.prototype._isInvalidEventTokenResponse = function(httpResponse, body) {
  if(!httpResponse || httpResponse.statusCode !== 401) return false;
  try {
    const response = JSON.parse(body || '{}');
    return response && response.payload && response.payload.code === 'INVALID_ACCESS_TOKEN_EXCEPTION';
  } catch(error) {
    return /INVALID_ACCESS_TOKEN_EXCEPTION/.test(String(body || ''));
  }
}

Server.prototype._applyEventToken = function(event) {
  if(event && event.event && event.event.endpoint && event.event.endpoint.scope)
    event.event.endpoint.scope.token = event_token.access_token;
  else if(event && event.event && event.event.payload && event.event.payload.endpoints &&
          event.event.payload.endpoints[0] && event.event.payload.endpoints[0].scope)
    event.event.payload.endpoints[0].scope.token = event_token.access_token;
  else if(event && event.directive && event.directive.endpoint && event.directive.endpoint.scope)
    event.directive.endpoint.scope.token = event_token.access_token;
}

Server.prototype._handleEventPostResult = function(event, err, httpResponse, body) {
  if(err) {
    log.error('failed to post event: network error');
    return;
  }
  const statusCode = httpResponse ? httpResponse.statusCode : 'unknown';
  log.info('posted skill event: '+ statusCode);
  if(this._isInvalidEventTokenResponse(httpResponse, body))
    this._recoverInvalidEventToken(event);
}

Server.prototype._recoverInvalidEventToken = function(event) {
  if(this._eventRecoveryStopped) return Promise.resolve(false);
  if(this._eventRecoveryPromise) return this._eventRecoveryPromise;

  const now = this._now();
  if(now < this._eventRecoveryCooldownUntil) {
    this._setEventRecoveryReading(RECOVERY_RESULT_READING, 'cooldown_active');
    return Promise.resolve(false);
  }

  this._eventRecoveryCooldownUntil = now + EVENT_RECOVERY_COOLDOWN;
  this._eventRecoveryCount++;
  this._setEventRecoveryReading(EVENT_ERROR_READING, 'invalid_access_token');
  this._setEventRecoveryReading(RECOVERY_REASON_READING, 'token_invalid');
  this._setEventRecoveryReading(RECOVERY_TIME_READING, new Date(now).toISOString());
  this._setEventRecoveryReading(RECOVERY_COUNT_READING, this._eventRecoveryCount);
  this._setEventRecoveryReading(RECOVERY_RESULT_READING, 'refreshing_token');

  const restartTunnel = () => {
    if(this._eventRecoveryStopped) return false;
    this._setEventRecoveryReading(RECOVERY_RESULT_READING, 'tunnel_restart_requested');
    this._restartSshTunnel();
    return false;
  };

  const recovery = new Promise(resolve => {
    event_token.expires_in = 0;
    this.updateEventToken(undefined, () => {
      if(this._eventRecoveryStopped) { resolve(false); return; }
      this._setEventRecoveryReading(RECOVERY_RESULT_READING, 'retrying_event');
      this._applyEventToken(event);
      this._sendEvent(event, (err, httpResponse, body) => {
        if(this._eventRecoveryStopped) { resolve(false); return; }
        if(!err && httpResponse && httpResponse.statusCode >= 200 && httpResponse.statusCode < 300) {
          this._setEventRecoveryReading(RECOVERY_RESULT_READING, 'successful');
          resolve(true);
        } else if(httpResponse && httpResponse.statusCode === 401) {
          resolve(restartTunnel());
        } else {
          this._setEventRecoveryReading(RECOVERY_RESULT_READING, 'event_retry_failed');
          resolve(false);
        }
      });
    }, () => resolve(restartTunnel()));
  });
  this._eventRecoveryPromise = recovery;
  recovery.then(() => {
    if(this._eventRecoveryPromise === recovery) this._eventRecoveryPromise = undefined;
  });
  return recovery;
}

Server.prototype.refreshToken = function(token, callback, errorCallback) {
  var permissions = {};
  var url = 'https://api.amazon.com/auth/O2/token';
  if( !this._config.alexa || !this._config.alexa.permissions ) {
    if( this._config.sshproxy ) {
      url = 'https://va.fhem.de/proxy/oauth';
    } else {
      log.error('no permissions in config.json');
      return;
    }
  } else
    permissions = this._config.alexa.permissions[Object.keys(this._config.alexa.permissions)[0]];

  log.info( 'refreshing token' );
  //var request = require('request');
  //require('request-debug')(request);
  //request.post(url,
  require('postman-request').post(url,
                          { form :{ 'grant_type': 'refresh_token',
                                    'refresh_token': token.refresh_token,
                                    'client_id': permissions.client_id?permissions.client_id:'',
                                    'client_secret': permissions.client_secret?permissions.client_secret:'' } },
                          function(err, httpResponse, body) {
                            if( err ) {
                              log.error( 'failed to refresh token: network error');
                              if( errorCallback ) errorCallback(err);
                              return;
                            }

                            const contentType = httpResponse.headers['content-type'];
                            if( contentType && contentType.match('application/json') ) {
                              var json = JSON.parse(body);
                              if( json.error ) {
                                log.error( 'failed to refresh token: '+ json.error);
                                if( errorCallback ) errorCallback(new Error(json.error));
                                return;
                              }

                              log.info( 'got fresh token' );
                              json.expires_in = json.expires_in * 1000 + Date.now();
                              for( let key in json )
                                token[key] = json[key];

                              if( callback ) callback(token);

                            } else {
                              log.error( 'failed to refresh token: invalid response');
                              if( errorCallback ) errorCallback(new Error('invalid token refresh response'));
                              return;
                            }

                          }.bind(this) );
}

var event_token = {'expires': 0};
Server.prototype.updateEventToken = function(grant,callback,errorCallback) {
  if( Date.now() < event_token.expires_in ) {
    if( callback ) callback();

  } else if( event_token.refresh_token ) {
    this.refreshToken(event_token, function(token) { event_token = token; if( callback ) callback(); }.bind(this), errorCallback );

  } else {
    if( !grant ) {
      log.error('no grant for event token update');
      if( errorCallback ) errorCallback(new Error('no grant for event token update'));
      return;
    }

    var permissions = {};
    var url = 'https://api.amazon.com/auth/O2/token';
    if( !this._config.alexa || !this._config.alexa.permissions ) {
      if( this._config.sshproxy ) {
        url = 'https://va.fhem.de/proxy/oauth';
      } else {
        log.error('no permissions in config.json');
        return;
      }
    } else
      permissions = this._config.alexa.permissions[Object.keys(this._config.alexa.permissions)[0]];

    log.info( 'requesting event token' );
    //log.error( grant );
    //var request = require('request');
    //require('request-debug')(request);
    //request.post(url,
    require('postman-request').post(url,
                            { form :{ 'grant_type': 'authorization_code',
                                      'code': grant.grant.code,
                                      'client_id': permissions.client_id?permissions.client_id:'',
                                      'client_secret': permissions.client_secret?permissions.client_secret:'' } },
                            function(err, httpResponse, body) {
                              if( err ) {
                              log.error( 'failed to get event token: network error');
                              if( errorCallback ) errorCallback(err);
                                return;
                              }

                              const contentType = httpResponse.headers['content-type'];
                              if( contentType && contentType.match('application/json') ) {
                                var json = JSON.parse(body);
                                if( json.error ) {
                                log.error( 'failed to get event token: '+ json.error);
                                if( errorCallback ) errorCallback(new Error(json.error));
                                  return;
                                }

                                this.setreading( '.eventToken', body );

                                log.info( 'got event token' );
                                event_token = json;
                                event_token.expires_in = event_token.expires_in * 1000 + Date.now();

                                if( callback ) callback();

                              } else {
                              log.error( 'failed to get event token: invalid response');
                              if( errorCallback ) errorCallback(new Error('invalid event token response'));
                                return;
                              }

                            }.bind(this) );
  }
}
